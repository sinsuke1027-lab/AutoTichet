```typescript
    } else {
      // 別カラムへのドロップ → 楽観的更新でステータスを即時反映
      const queryKey = ['tasks-view', { project_id: projectId }]

      // 現在のキャッシュを保存
      const previous = queryClient.getQueryData<Task[]>(queryKey)

      // キャッシュを即時更新
      queryClient.setQueryData<Task[]>(queryKey, (old = []) =>
        old.map((t) => (t.id === activeId ? { ...t, status: overColKey } : t))
      )

      // API リクエスト（失敗時にロールバック）
      updateTask.mutate(
        { id: activeId, status: overColKey },
        {
          onError: () => {
            queryClient.setQueryData(queryKey, previous)
          },
          onSettled: () => {
            queryClient.invalidateQueries({ queryKey: ['tasks-view'] })
          },
        }
      )
    }
```
